help_title = 📚 epubcreator — 将目录编译为 EPUB 的 CLI 工具 (v{version})
help_commands = 
  node epubcreator.js createdir           创建目录结构并下载工具配置
  node epubcreator.js convertch           将 .md → .xhtml（默认，在 Markdowns/ 中查找）
  node epubcreator.js convertch xhtml2md  将 .xhtml → .md
  node epubcreator.js convertch docx2md   将 .docx → .md（按 ## 拆分）
  node epubcreator.js conv ...            convertch 的别名
  node epubcreator.js split <路径>        按标题 ## 拆分 .md 文件
  node epubcreator.js merge <路径>        将 .md 文件合并为一个（拆分的逆操作）
  node epubcreator.js build               从当前目录构建 EPUB
  node epubcreator.js debug               监视 Markdowns/ 中的更改，自动 convertch + build
  node epubcreator.js import              从当前目录导入 .epub 文件
  node epubcreator.js updatemodule        从仓库下载/更新 node_modules
  node epubcreator.js updatemodule --force 无需确认更新
  node epubcreator.js updateconfig        从仓库（config 分支）下载/更新 .epubcreator
  node epubcreator.js settings            交互式工具设置
  node epubcreator.js validate            使用 epubcheck 验证最新的 EPUB
  node epubcreator.js --version           显示版本

  │   ├── audiovideo/     ← 可选
  │   ├── bab1.xhtml      ← 章节（任意名称，在 EPUB/ 根目录下）
